# Standalone, offline verification uses only owned fixture code and data checks.
cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(sf-semantic-attribution-offline LANGUAGES CXX ASM_MASM)
    enable_testing()
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Semantic attribution requires Windows x64 MSVC/MASM")
endif()
get_filename_component(native_dir "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
add_library(sf-attribution-native-adapters OBJECT EXCLUDE_FROM_ALL
    native_site_profile.cpp native_cpu_state.cpp
    native_frame_capture.cpp native_event_adapter.cpp native_thunks.asm prepare_attribution.cpp)
add_library(sf-attribution-native STATIC EXCLUDE_FROM_ALL owned_ledger.cpp $<TARGET_OBJECTS:sf-attribution-native-adapters>)
foreach(t sf-attribution-native sf-attribution-native-adapters)
    target_include_directories(${t} PRIVATE "${native_dir}")
    target_compile_features(${t} PRIVATE cxx_std_20)
    target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${t} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/EHsc>")
    set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
add_executable(sf-attribution-relay-data EXCLUDE_FROM_ALL
    native_site_profile.cpp native_relay_tests.cpp)
add_executable(sf-attribution-thunk-fixture EXCLUDE_FROM_ALL
    native_site_profile.cpp native_cpu_state.cpp native_frame_capture.cpp native_thunks.asm native_thunk_fixture.asm native_thunk_tests.cpp)
add_executable(sf-attribution-owned-ledger EXCLUDE_FROM_ALL owned_ledger.cpp owned_ledger_tests.cpp)
add_executable(sf-attribution-close-drain-fixture EXCLUDE_FROM_ALL
    native_adapter_concurrency_tests.cpp native_event_adapter.cpp native_frame_capture.cpp native_site_profile.cpp owned_ledger.cpp)
target_compile_definitions(sf-attribution-close-drain-fixture PRIVATE SF_SEMANTIC_ATTRIBUTION_TESTING)
target_compile_definitions(sf-attribution-owned-ledger PRIVATE SF_SEMANTIC_ATTRIBUTION_TESTING)
add_executable(sf-attribution-patch-set-fixture EXCLUDE_FROM_ALL "${native_dir}/../tests/executable_write_set_test.cpp")
foreach(t sf-attribution-relay-data sf-attribution-thunk-fixture sf-attribution-owned-ledger sf-attribution-patch-set-fixture sf-attribution-close-drain-fixture)
    target_include_directories(${t} PRIVATE "${native_dir}")
    target_compile_features(${t} PRIVATE cxx_std_20)
    target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${t} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/EHsc;/UNDEBUG>")
    set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_test(NAME ${t} COMMAND ${t})
endforeach()
set_tests_properties(sf-attribution-thunk-fixture PROPERTIES SKIP_RETURN_CODE 77)
